    <!-- ADAPT-2026-07-31.
         `maximum-scale=1, user-scalable=no` removed: it disabled pinch-zoom
         app-wide, which is a WCAG 1.4.4 (AA) failure — low-vision users had no
         magnification path at all, and the in-app --font-scale control is not a
         substitute for viewport zoom. iOS auto-zooms a focused input under 16px,
         which is the reason this flag usually gets added; that is handled
         properly by the (pointer: coarse) font-size floor at the end of App.css.
         X03: the chat composer is exempt from that floor (it is the bubble's
         14px); on iOS the script right below stops its focus zoom instead.

         `viewport-fit=cover` added: without it env(safe-area-inset-*) resolves
         to 0, so the 14 safe-area rules already written across this codebase —
         including .navContainer's home-indicator lift — were dead code. This
         entry declares apple-mobile-web-app-status-bar-style: black-translucent,
         so an installed PWA already drew under the status bar with nothing
         compensating for it. -->
    <meta name="viewport" content="width=device-width, initial-scale=1, viewport-fit=cover, interactive-widget=resizes-content">
    <script>
      // X03: iOS only. Since iOS 10 every iOS browser ignores maximum-scale for
      // the user's own pinch, so this is NOT a zoom lock there — it only stops
      // the automatic zoom when a field under 16px takes focus (the composer).
      // Android honours it as a lock, which is why it never runs there. Same
      // test as App.jsx isIosDevice (iPadOS 13+ reports itself as a Mac).
      (function () {
        var ua = navigator.userAgent || '';
        if (!(/iphone|ipad|ipod/i.test(ua) || (/macintosh/i.test(ua) && navigator.maxTouchPoints > 1))) return;
        var m = document.querySelector('meta[name="viewport"]');
        if (m) m.setAttribute('content', m.getAttribute('content') + ', maximum-scale=1');
      })();
    </script>
